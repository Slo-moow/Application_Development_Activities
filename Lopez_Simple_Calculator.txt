package com.example.lopez_activity;

import android.annotation.SuppressLint;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;
import androidx.core.view.WindowCompat;
import androidx.appcompat.app.AppCompatActivity;

import kotlin.TuplesKt;


public class MainActivity extends AppCompatActivity {
    EditText fnum, snum;

    Button btadd;
    Button btsub;
    Button btmul;
    Button btdiv;
    Button bteql;

    TextView txtview;

    @SuppressLint("MissingInflatedId")
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        WindowCompat.enableEdgeToEdge(getWindow());
        setContentView(R.layout.activity_main);

        Button btadd = (Button) findViewById(R.id.btadd);
        Button btsub = (Button) findViewById(R.id.btsub);
        Button btmul = (Button) findViewById(R.id.btmul);
        Button btdiv = (Button) findViewById(R.id.btdiv);
        Button bteql = (Button) findViewById(R.id.bteql);
        fnum = findViewById(R.id.fnum);
        snum = findViewById(R.id.fnum);
        txtview = findViewById(R.id.txtView);


      btadd.setOnClickListener(new View.OnClickListener() {
          @Override public void onClick(View view) {
              String fnumber = fnum.getText().toString();
              String snumber = snum.getText().toString();
              if(fnumber.isEmpty()) {
                  Toast.makeText(MainActivity.this, "Input 1st number",Toast.LENGTH_SHORT).show(); }
              else if (snumber.isEmpty()) {
                  Toast.makeText(MainActivity.this,"Input 2nd  number",Toast.LENGTH_SHORT).show(); }
              else {
                  int num1 = Integer.parseInt(fnum.getText().toString());
                  int num2 = Integer.parseInt(snum.getText().toString());
                  int sum = Math.addExact(num1,num2);

                  txtview.setText(String.valueOf(sum));
              }
           }
        });

      btsub.setOnClickListener(new View.OnClickListener() {
          @Override public void onClick(View view) {
              String fnumber = fnum.getText().toString();
              String snumber = snum.getText().toString();
              if(fnumber.isEmpty()) {
                  Toast.makeText(MainActivity.this, "Input 1st number",Toast.LENGTH_SHORT).show(); }
              else if (snumber.isEmpty()) {
                  Toast.makeText(MainActivity.this,"Input 2nd  number",Toast.LENGTH_SHORT).show(); }
              else {
                  int num1 = Integer.parseInt(fnum.getText().toString());
                  int num2 = Integer.parseInt(snum.getText().toString());
                  int sub = Math.subtractExact(num1,num2);

                  txtview.setText(String.valueOf(sub));
              }

          }
      });

      btmul.setOnClickListener(new View.OnClickListener() {
          @Override public void onClick(View v) {
              String fnumber = fnum.getText().toString();
              String snumber = snum.getText().toString();
              if (fnumber.isEmpty()) {
                  Toast.makeText(MainActivity.this, "Input 1st number", Toast.LENGTH_SHORT).show();
              } else if (snumber.isEmpty()) {
                  Toast.makeText(MainActivity.this, "Input 2nd  number", Toast.LENGTH_SHORT).show();
              } else {
                  int num1 = Integer.parseInt(fnum.getText().toString());
                  int num2 = Integer.parseInt(snum.getText().toString());
                  int mul = Math.multiplyExact(num1, num2);

                  txtview.setText(String.valueOf(mul));
              }
          }
      });

      btdiv.setOnClickListener(new View.OnClickListener() {
          @Override
          public void onClick(View v) {
              String fnumber = fnum.getText().toString();
              String snumber = snum.getText().toString();
              if (fnumber.isEmpty()) {
                  Toast.makeText(MainActivity.this, "Input 1st number", Toast.LENGTH_SHORT).show();
              } else if (snumber.isEmpty()) {
                  Toast.makeText(MainActivity.this, "Input 2nd  number", Toast.LENGTH_SHORT).show();
              } else {
                  int num1 = Integer.parseInt(fnum.getText().toString());
                  int num2 = Integer.parseInt(snum.getText().toString());
                  int div = Math.subtractExact(num1, num2);

                  txtview.setText(String.valueOf(div));

              }
          }
      });

    }
}
